-- Safe ScreenGui Parent Function
local function GetSafeParent()
    local parent
    pcall(function()
        if gethui then
            parent = gethui()
        elseif syn and syn.protect_gui then
            parent = game:GetService("CoreGui")
            syn.protect_gui(parent)
        elseif game:GetService("CoreGui") then
            parent = game:GetService("CoreGui")
        end
    end)
    if not parent then
        parent = LocalPlayer:WaitForChild("PlayerGui")
    end
    return parent
end

    SliderBtn.Position = UDim2.new(0.05, 0, 0.65, 0)
    SliderBtn.BackgroundColor3 = Color3.fromRGB(30, 42, 58)
    SliderBtn.Text = ""

    local SliderCorner = Instance.new("UICorner", SliderBtn)
    SliderCorner.CornerRadius = UDim.new(1, 0)

    local Fill = Instance.new("Frame", SliderBtn)
    Fill.Size = UDim2.new((Config[key] - min)/(max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
    Fill.BorderSizePixel = 0

    local FillCorner = Instance.new("UICorner", Fill)
    FillCorner.CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function Update(input)
        local pos = math.clamp((input.Position.X - SliderBtn.AbsolutePosition.X) / SliderBtn.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + ((max - min) * pos))
        Fill.Size = UDim2.new(pos, 0, 1, 0)
        Label.Text = text .. ": " .. tostring(val)
        Config[key] = val
        if callback then callback(val) end
    end

    SliderBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            Update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            Update(input)
        end
    end)
end

-- TARGETING & TEAM FILTER LOGIC
---------------------------------------------------------
local function IsValidTarget(player)
    if not player or player == LocalPlayer or not player.Character then return false end
    
    local hum = player.Character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end

    if Config.TeamCheck then
        local myTeam = tostring(LocalPlayer.Team)
        local targetTeam = tostring(player.Team)
        
        if Config.TargetTeam == "Guards" and targetTeam ~= "Guards" then return false end
        if Config.TargetTeam == "Inmates" and targetTeam ~= "Inmates" then return false end
        if Config.TargetTeam == "Criminals" and targetTeam ~= "Criminals" then return false end
        if Config.TargetTeam ~= "All" and myTeam == targetTeam then return false end
    end

    return true
end

local function GetFOVCenter()
    if Config.FOV_Mode == "Static" then
        return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    else
        return UserInputService:GetMouseLocation()
    end
end

local function GetClosestPlayer()
    local closestPart = nil
    local shortestDistance = Config.FOV_Size
    local center = GetFOVCenter()

    for _, player in ipairs(Players:GetPlayers()) do
        if IsValidTarget(player) and player.Character:FindFirstChild(Config.AimTarget) then
            local part = player.Character[Config.AimTarget]
            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)

            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if dist < shortestDistance then
                    shortestDistance = dist
                    closestPart = part
                end
            end
        end
    end
    return closestPart
end

---------------------------------------------------------
-- SILENT AIM HOOK (PRISON LIFE RAYCAST INTERCEPT)
---------------------------------------------------------
pcall(function()
    local rawMetatable = getrawmetatable or debug.getmetatable
    local setReadOnly = setreadonly or make_writeable
    if rawMetatable and setReadOnly then
        local gmt = rawMetatable(game)
        local oldNamecall = gmt.__namecall
        setReadOnly(gmt, false)

        gmt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = {...}

            if Config.SilentAim and tostring(method) == "Raycast" then
                local targetPart = GetClosestPlayer()
                if targetPart then
                    local origin = args[1]
                    local newDirection = (targetPart.Position - origin).Unit * 1000
                    args[2] = newDirection
                    return oldNamecall(self, unpack(args))
                end
            end

            return oldNamecall(self, ...)
        end)
        setReadOnly(gmt, true)
    end
end)

local angle = 0
local jitterState = false

RunService.RenderStepped:Connect(function(deltaTime)
    -- FOV Update
    if FOVCircle then
        FOVCircle.Visible = Config.FOV_Visible
        FOVCircle.Radius = Config.FOV_Size
        FOVCircle.Color = Config.FOV_Color
        FOVCircle.Position = GetFOVCenter()
    end

    -- Visuals Update
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            
            -- Chams color check
            local teamColor = player.TeamColor and player.TeamColor.Color or Config.ChamsColor
            local activeChamsColor = Config.TeamColorChams and teamColor or Config.ChamsColor

            local hl = char:FindFirstChild("GxiHighlightNL")
            if hl then
                hl.Enabled = Config.ESP
                hl.FillColor = Config.ESP_Color
            end

            for _, child in ipairs(char:GetChildren()) do
                if child:IsA("BoxHandleAdornment") and child.Name == "GxiChamsNL" then
                    child.Visible = Config.Chams
                    child.Color3 = activeChamsColor
                end
            end

            local head = char:FindFirstChild("Head")
            if head then
                local bbg = head:FindFirstChild("GxiBillboardNL")
                if bbg then
                    local nameLabel = bbg:FindFirstChild("NameLabel")
                    if nameLabel then
                        nameLabel.Visible = Config.Names
                        nameLabel.TextSize = Config.NameSize
                    end

                    local hpBg = bbg:FindFirstChild("HPBg")
                    if hpBg then
                        hpBg.Visible = Config.HPBar
                        local hpFill = hpBg:FindFirstChild("HPFill")
                        local hum = char:FindFirstChildOfClass("Humanoid")
                        if hpFill and hum then
                            local healthPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                            hpFill.Size = UDim2.new(healthPct, 0, 1, 0)
                        end
                    end
                end
            end
        end
    end

    -- Aimbot Smooth Aim Loop
    if Config.Aimbot then
        local targetPart = GetClosestPlayer()
        if targetPart then
            local targetCamPos = CFrame.new(Camera.CFrame.Position, targetPart.Position)
            local smoothnessFactor = math.clamp((Config.AimSpeed / 100) * (deltaTime * 60), 0.01, 1)
            Camera.CFrame = Camera.CFrame:Lerp(targetCamPos, smoothnessFactor)
        end
    end

    -- Anti-Aim (Jitter / Spin)
    if Config.AntiAim and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local hrp = LocalPlayer.Character.HumanoidRootPart
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        
        angle = (angle + Config.AntiAimSpeed * deltaTime * 10) % 360

        if hum then
            local isMoving = hum.MoveDirection.Magnitude > 0
            if not isMoving then
                hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, math.rad(angle), 0)
            else
                jitterState = not jitterState
                local jitterAngle = jitterState and 45 or -45
                local moveDir = hum.MoveDirection
                local backLook = CFrame.lookAt(hrp.Position, hrp.Position - moveDir)
                hrp.CFrame = backLook * CFrame.Angles(0, math.rad(jitterAngle), 0)
            end
        end
    end
