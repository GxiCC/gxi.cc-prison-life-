-- Safe ScreenGui Parent Function
local function GetSafeParent()
    local parent
    pcall(function()
        if gethui then
            parent = gethui()
        elseif syn and syn.protect_gui then
            parent = game:GetService("CoreGui")
            syn.protect_gui(parent)
        elseif game:GetService("CoreGui") then
            parent = game:GetService("CoreGui")
        end
    end)
    if not parent then
        parent = LocalPlayer:WaitForChild("PlayerGui")
    end
    return parent
end

local function CreateTab(tabName, layoutOrder)
    local TabContainer = Instance.new("ScrollingFrame")
    TabContainer.Name = tabName .. "Tab"
    TabContainer.Parent = ContentArea
    TabContainer.Size = UDim2.new(1, 0, 1, 0)
    TabContainer.BackgroundTransparency = 1
    TabContainer.BorderSizePixel = 0
    TabContainer.ScrollBarThickness = 4
    TabContainer.ScrollBarImageColor3 = Color3.fromRGB(255, 105, 180)
    TabContainer.Visible = false
    TabContainer.CanvasSize = UDim2.new(0, 0, 0, 0) -- Auto calculated

    local TabList = Instance.new("UIListLayout", TabContainer)
    TabList.SortOrder = Enum.SortOrder.LayoutOrder
    TabList.Padding = UDim.new(0, 10)
    TabList.HorizontalAlignment = Enum.HorizontalAlignment.Center

    TabList:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        TabContainer.CanvasSize = UDim2.new(0, 0, 0, TabList.AbsoluteContentSize.Y + 10)
    end)

    Tabs[tabName] = TabContainer

    -- Sidebar Button
    local TabBtn = Instance.new("TextButton")
    TabBtn.Name = tabName .. "Btn"
    TabBtn.Parent = Sidebar
    TabBtn.Size = UDim2.new(0.9, 0, 0, 32)
    TabBtn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
    TabBtn.TextColor3 = Color3.fromRGB(180, 180, 190)
    TabBtn.Text = tabName:upper()
    TabBtn.TextSize = 13
    TabBtn.Font = Enum.Font.SourceSansBold
    TabBtn.LayoutOrder = layoutOrder

    local BtnCorner = Instance.new("UICorner", TabBtn)
    BtnCorner.CornerRadius = UDim.new(0, 6)

    TabButtons[tabName] = TabBtn

    TabBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            for name, container in pairs(Tabs) do
                container.Visible = (name == tabName)
            end
            for name, btn in pairs(TabButtons) do
                if name == tabName then
                    btn.BackgroundColor3 = Color3.fromRGB(255, 105, 180)
                    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
                else
                    btn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
                    btn.TextColor3 = Color3.fromRGB(180, 180, 190)
                end
            end
        end
    end)

    return TabContainer
end

    Label.BackgroundTransparency = 1
    Label.Text = text
    Label.TextColor3 = Color3.fromRGB(220, 220, 220)

    Label.Font = Enum.Font.SourceSansSemibold

    local SwitchBg = Instance.new("Frame")
    SwitchBg.Parent = Container

    SwitchBg.Position = UDim2.new(0.95, -40, 0.5, -10)

    local SwitchCorner = Instance.new("UICorner", SwitchBg)
    SwitchCorner.CornerRadius = UDim.new(1, 0)

    local Knob = Instance.new("Frame")
    Knob.Parent = SwitchBg
    Knob.Size = UDim2.new(0, 16, 0, 16)
    Knob.Position = Config[key] and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
    Knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

    local KnobCorner = Instance.new("UICorner", Knob)
    KnobCorner.CornerRadius = UDim.new(1, 0)

    local ClickBtn = Instance.new("TextButton")
    ClickBtn.Parent = Container
    ClickBtn.Size = UDim2.new(1, 0, 1, 0)
    ClickBtn.BackgroundTransparency = 1
    ClickBtn.Text = ""

    local function updateVisual(state)
        local targetBg = state and Color3.fromRGB(255, 105, 180) or Color3.fromRGB(45, 45, 55)
        local targetPos = state and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
        TweenService:Create(SwitchBg, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = targetBg}):Play()
        TweenService:Create(Knob, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = targetPos}):Play()
    end

    ClickBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Config[key] = not Config[key]
            updateVisual(Config[key])
            if callback then callback(Config[key]) end
        end
    end)
end

local function CreateSlider(column, text, min, max, key, layoutOrder, callback)
    local Frame = Instance.new("Frame")
    Frame.Parent = column
    Frame.Size = UDim2.new(0.92, 0, 0, 40)
    Frame.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
    Frame.LayoutOrder = layoutOrder

    local Corner = Instance.new("UICorner", Frame)
    Corner.CornerRadius = UDim.new(0, 6)

    local Label = Instance.new("TextLabel")
    Label.Parent = Frame
    Label.Size = UDim2.new(1, 0, 0, 18)
    Label.BackgroundTransparency = 1
    Label.Text = text .. ": " .. tostring(Config[key])
    Label.TextColor3 = Color3.fromRGB(220, 220, 220)
    Label.TextSize = 12
    Label.Font = Enum.Font.SourceSans

    local SliderBtn = Instance.new("TextButton")
    SliderBtn.Parent = Frame
    SliderBtn.Size = UDim2.new(0.9, 0, 0, 6)
    SliderBtn.Position = UDim2.new(0.05, 0, 0.68, 0)
    SliderBtn.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
    SliderBtn.Text = ""

    local SliderCorner = Instance.new("UICorner", SliderBtn)
    SliderCorner.CornerRadius = UDim.new(1, 0)

    local Fill = Instance.new("Frame")
    Fill.Parent = SliderBtn
    Fill.Size = UDim2.new((Config[key] - min)/(max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(255, 105, 180)
    Fill.BorderSizePixel = 0

    local FillCorner = Instance.new("UICorner", Fill)
    FillCorner.CornerRadius = UDim.new(1, 0)

    local dragging = false
    local function Update(input)
        local pos = math.clamp((input.Position.X - SliderBtn.AbsolutePosition.X) / SliderBtn.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + ((max - min) * pos))
        Fill.Size = UDim2.new(pos, 0, 1, 0)
        Label.Text = text .. ": " .. tostring(val)
        Config[key] = val
        if callback then callback(val) end
    end

    SliderBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            Update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            Update(input)
        end
    end)
end

local function CreateSelector(column, text, options, key, layoutOrder, callback)
    local Btn = Instance.new("TextButton")
    Btn.Parent = column
    Btn.Size = UDim2.new(0.92, 0, 0, 32)
    Btn.BackgroundColor3 = Color3.fromRGB(28, 28, 35)
    Btn.TextColor3 = Color3.fromRGB(255, 182, 193)
    Btn.Text = text .. ": " .. tostring(Config[key])
    Btn.TextSize = 12
    Btn.Font = Enum.Font.SourceSansBold
    Btn.LayoutOrder = layoutOrder

    local BtnCorner = Instance.new("UICorner", Btn)
    BtnCorner.CornerRadius = UDim.new(0, 6)

    Btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local currentIndex = 1
            for i, opt in ipairs(options) do
                if opt == Config[key] then currentIndex = i break end
            end
            local nextIndex = currentIndex % #options + 1
            Config[key] = options[nextIndex]
            Btn.Text = text .. ": " .. tostring(Config[key])
            if callback then callback(Config[key]) end
        end
    end)
end

                nameLabel.TextSize = Config.NameSize
                nameLabel.Font = Enum.Font.SourceSansBold

                hpFill.BackgroundColor3 = Color3.fromRGB(0, 255, 120)

local function MonitorHealth(player)
    if player == LocalPlayer then return end

    local function TrackChar(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if hum then
            hum.Died:Connect(function()
                if Config.Aimbot or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head")) then
                    local myPos = LocalPlayer.Character.Head.Position
                    local enemyPos = char.Head and char.Head.Position or myPos
                    if (myPos - enemyPos).Magnitude < 250 then
                        TriggerKillEffects(player.DisplayName)
                    end
                end
            end)
        end
    end

    if player.Character then TrackChar(player.Character) end
    player.CharacterAdded:Connect(TrackChar)
end

for _, plr in ipairs(Players:GetPlayers()) do MonitorHealth(plr) end
Players.PlayerAdded:Connect(MonitorHealth)

local angle = 0
local jitterState = false

local function GetFOVCenter()
    if Config.FOV_Mode == "Static" then
        return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    else
        return UserInputService:GetMouseLocation()
    end
end

local function GetClosestPlayer()
    local closestPlayer = nil
    local shortestDistance = Config.FOV_Size
    local center = GetFOVCenter()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild(Config.AimTarget) then
            local part = player.Character[Config.AimTarget]
            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)

            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if dist < shortestDistance then
                    shortestDistance = dist
                    closestPlayer = part
                end
            end
        end
    end
    return closestPlayer
end

RunService.RenderStepped:Connect(function(deltaTime)
    -- FOV Update
    if FOVCircle then
        FOVCircle.Visible = Config.FOV_Visible
        FOVCircle.Radius = Config.FOV_Size
        FOVCircle.Color = Config.FOV_Color
        FOVCircle.Position = GetFOVCenter()
    end

    -- Visuals Update
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local char = player.Character
            
            local hl = char:FindFirstChild("GxiHighlight")
            if hl then
                hl.Enabled = Config.ESP
                hl.FillColor = Config.ESP_Color
            end

            for _, child in ipairs(char:GetChildren()) do
                if child:IsA("BoxHandleAdornment") and child.Name == "GxiChams" then
                    child.Visible = Config.Chams
                    child.Color3 = Config.ChamsColor
                end
            end

            local head = char:FindFirstChild("Head")
            if head then
                local bbg = head:FindFirstChild("GxiBillboard")
                if bbg then
                    local nameLabel = bbg:FindFirstChild("NameLabel")
                    if nameLabel then
                        nameLabel.Visible = Config.Names
                        nameLabel.TextSize = Config.NameSize
                    end

                    local hpBg = bbg:FindFirstChild("HPBg")
                    if hpBg then
                        hpBg.Visible = Config.HPBar
                        local hpFill = hpBg:FindFirstChild("HPFill")
                        local hum = char:FindFirstChildOfClass("Humanoid")
                        if hpFill and hum then
                            local healthPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                            hpFill.Size = UDim2.new(healthPct, 0, 1, 0)
                            if healthPct > 0.5 then
                                hpFill.BackgroundColor3 = Color3.fromRGB(0, 255, 120)
                            elseif healthPct > 0.2 then
                                hpFill.BackgroundColor3 = Color3.fromRGB(255, 180, 0)
                            else
                                hpFill.BackgroundColor3 = Color3.fromRGB(255, 40, 40)
                            end
                        end
                    end
                end
            end
        end
    end

    -- Aimbot Update
    if Config.Aimbot then
        local targetPart = GetClosestPlayer()
        if targetPart then
            local targetCamPos = CFrame.new(Camera.CFrame.Position, targetPart.Position)
            Camera.CFrame = Camera.CFrame:Lerp(targetCamPos, Config.Smoothness)
        end
    end

    -- Anti-Aim Update
    if Config.AntiAim and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        local hrp = LocalPlayer.Character.HumanoidRootPart
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        
        angle = (angle + Config.AntiAimSpeed * deltaTime * 10) % 360

        if hum then
            local isMoving = hum.MoveDirection.Magnitude > 0
            local isJumping = hum:GetState() == Enum.HumanoidStateType.Freefall or hum:GetState() == Enum.HumanoidStateType.Jumping

            if isJumping or not isMoving then
                hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, math.rad(angle), 0)
            elseif isMoving then
                jitterState = not jitterState
                local jitterAngle = jitterState and 45 or -45
                local moveDir = hum.MoveDirection
                local backLook = CFrame.lookAt(hrp.Position, hrp.Position - moveDir)
                hrp.CFrame = backLook * CFrame.Angles(0, math.rad(jitterAngle), 0)
            end
        end
    end
end)
